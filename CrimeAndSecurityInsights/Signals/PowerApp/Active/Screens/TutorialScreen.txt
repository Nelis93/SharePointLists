                  SharePointForm.Mode <> FormMode.View,
                      Navigate(RenderingScreen,ScreenTransition.None);
                      Set(isTutorialRendered,false),

                  ClearCollect(Topics, SharePointIntegration.Selected.Topic);
                  ClearCollect(otherTopics,[]);
                  ClearCollect(Status2, SharePointIntegration.Selected.'Status (2)');

                  Set(isRecipient, CountRows(Filter(SharePointIntegration.Selected.Recipients,Email = User().Email))>0);
                  Set(isAuthor, SharePointIntegration.Selected.'Gemaakt door'.Email = User().Email);
                  Set(isTeamMember, CountRows(Filter(SharePointIntegration.Selected.Team,Email = User().Email))>0);
                  Set(isMember,Or(isEditor,isAuthor,isRecipient,isTeamMember));
                  Set(isForDirCom,CountRows(Filter(Status2, Value = "For DirCom"))>0);
                  Set(isPrioCalled,Or(CountRows(Filter(Status2, Value = "Confirmed by DirCom"))>0,CountRows(Filter(Status2, Value = "Noticed by DirCom"))>0));

                      And(isTeamMember,!isRecipient),

                              Validators,

                                      SharePointIntegration.Selected.Team,
                                      Lower(Email) in
                                          ForAll('Office365Groups'.ListGroupMembers(
                                              LookUp(
                                                  'Office365Groups'.ListOwnedGroups().value,
                                                  displayName = "Signals_TeamLeaders (Grp)"
                                              ).id
                                          ).value,

                                  )
                              )
                          )
                  );

                      isRecipient,

                              Validators,
                                  Filter(
                                      SharePointIntegration.Selected.Team,
                                      Lower(Email) in
                                          ForAll('Office365Groups'.ListGroupMembers(
                                              LookUp(
                                                  'Office365Groups'.ListOwnedGroups().value,
                                                  displayName = "Signals_CentrexLeaders (Grp)"
                                              ).id
                                          ).value, 
                                      Lower(mail)
                                  )
                              )
                          )
                  );
